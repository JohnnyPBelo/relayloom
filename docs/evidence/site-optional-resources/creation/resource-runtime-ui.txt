import {spawn,execFileSync} from 'node:child_process';
import {mkdirSync,readFileSync,writeFileSync,createWriteStream,statfsSync,copyFileSync} from 'node:fs';
import {resolve,join} from 'node:path';
import {createHash} from 'node:crypto';
const parent='/home/absint0o/projects/relayloom',out=resolve('.cache/resource-runtime-ui'),tmp=resolve('.cache/tmp');
mkdirSync(out,{recursive:true});mkdirSync(tmp,{recursive:true});
const previous=JSON.parse(readFileSync('.cache/resource-runtime-gate/report.json','utf8'));
if(previous.status!=='PASS')throw Error('API gate must finish successfully first');
for(const engine of ['chromium','firefox','webkit'])copyFileSync('.cache/browser-matrix/'+engine+'/playwright.json','.cache/resource-runtime-gate/'+engine+'.json');
const disk=statfsSync('.');if(disk.bavail*disk.bsize<15*1024**3)throw Error('disk reserve');
function hashes(){return Object.fromEntries(Object.keys(previous.sources).map(p=>[p,createHash('sha256').update(readFileSync(p)).digest('hex')]));}
if(JSON.stringify(hashes())!==JSON.stringify(previous.sources))throw Error('source changed since API gate');
const report={status:'RUNNING',scope:'Actual shared UI regression of sites; optional resource editor remains unimplemented',started:new Date().toISOString(),sources:previous.sources,checks:[]};
const save=()=>writeFileSync(join(out,'report.json'),JSON.stringify(report,null,2)+'\n');save();
const env={...process.env,TMPDIR:tmp,TMP:tmp,TEMP:tmp,PLAYWRIGHT_BROWSERS_PATH:parent+'/.cache/playwright',XDG_CACHE_HOME:resolve('.cache/runtime-cache')};
async function run(name,args,extra){const entry={name,command:[process.execPath,...args],started:new Date().toISOString(),status:'RUNNING'};report.checks.push(entry);save();console.log('START '+name);const log=createWriteStream(join(out,name+'.log'));const code=await new Promise((done,reject)=>{const child=spawn(process.execPath,args,{env:{...env,...extra},stdio:['ignore','pipe','pipe']});child.stdout.pipe(log,{end:false});child.stderr.pipe(log,{end:false});child.once('error',reject);child.once('exit',code=>log.end(()=>done(code)));});copyFileSync(name==='ui-node'||name==='ui-native'?'test-results/e2e.json':'.cache/browser-matrix/'+name.slice(3)+'/playwright.json',join(out,name+'.json'));entry.exitCode=code;entry.status=code===0?'PASS':'FAIL';save();console.log(entry.status+' '+name);if(code!==0)throw Error(name+' failed');}
try{
 for(const backend of ['node','native'])await run('ui-'+backend,[parent+'/node_modules/playwright/cli.js','test','--config','playwright.config.ts','tests/e2e/site-publication.spec.ts','--output',join(out,backend+'-artifacts')],{RELAYLOOM_TEST_BACKEND:backend});
 for(const engine of ['chromium','firefox','webkit'])await run('ui-'+engine,[parent+'/node_modules/playwright/cli.js','test','--config','tests/browser/matrix.config.ts','tests/browser/site-publication.spec.ts','--output',join(out,engine+'-artifacts')],{RELAYLOOM_MATRIX_ENGINE:engine});
 if(JSON.stringify(hashes())!==JSON.stringify(previous.sources))throw Error('source changed during UI gate');
 report.status='PASS';
}catch(error){report.status='FAIL';report.error=String(error);process.exitCode=1;}finally{report.finished=new Date().toISOString();save();}
